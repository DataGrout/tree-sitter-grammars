CREATE OR ALTER PROCEDURE [Sales].[CloseOrder] @OrderID int, @Note AS nvarchar(100) = NULL OUTPUT
WITH EXECUTE AS OWNER
AS
BEGIN
    SET NOCOUNT ON;
    UPDATE Sales.Orders SET ClosedWhen = SYSDATETIME() WHERE OrderID = @OrderID;
END;
GO
